def undistort(x, y, k, cx=P8_CX, cy=P8_CY):
    dx, dy = x - cx, y - cy
    s = 1 + k * (dx * dx + dy * dy)
    return cx + dx * s, cy + dy * s


# ---- The balusters' vertical vanishing point, refitted on distortion-corrected positions -----------------------------------------------
def load_balusters(path):
    tiers = {'U': [], 'M': []}
    for line in open(path, encoding='utf-8'):
        if not line.strip() or line.startswith('#'):
            continue
        t, i, x, y = line.split()
        tiers[t].append((int(i), float(x), float(y)))
    return tiers


def vanishing_point(samples, k, yref, cx=P8_CX, cy=P8_CY):
    """Fit a line x = a + b (y - yref) through each baluster's corrected centres, then the point they meet: a_i = X - (Y - yref) b_i."""
    by = {}
    for i, x, y in samples:
        by.setdefault(i, []).append(undistort(x, y, k, cx, cy))
    lines = []
    for pts in by.values():
        if len(pts) < 20:
            continue
        n = len(pts)
        sy = sum(p[1] - yref for p in pts); sx = sum(p[0] for p in pts)
        syy = sum((p[1] - yref) ** 2 for p in pts); sxy = sum(p[0] * (p[1] - yref) for p in pts)
        b = (n * sxy - sy * sx) / (n * syy - sy * sy)
        a = (sx - b * sy) / n
        rms = math.sqrt(sum((p[0] - (a + b * (p[1] - yref))) ** 2 for p in pts) / n)
        if rms < 2.5:
            lines.append((a, b, rms))
    m = len(lines)
    sb = sum(l[1] for l in lines); sa = sum(l[0] for l in lines)
    sbb = sum(l[1] ** 2 for l in lines); sab = sum(l[0] * l[1] for l in lines)
    slope = (m * sab - sb * sa) / (m * sbb - sb * sb)
    X = (sa - slope * sb) / m
    rms = math.sqrt(sum((l[0] - (X + slope * l[1])) ** 2 for l in lines) / m)
    line_rms = math.sqrt(sum(l[2] ** 2 for l in lines) / m)
    return X, yref - slope, m, rms, line_rms


def refit(path, cx=P8_CX, cy=P8_CY, show=False):
    tiers = load_balusters(path)
    grid = [j * 0.1e-8 for j in range(0, 51)]
    rows = []
    for k in grid:
        u = vanishing_point(tiers['U'], k, 820, cx, cy)
        m = vanishing_point(tiers['M'], k, 1350, cx, cy)
        rows.append((k, u, m))
        if show and abs(round(k * 1e9) % 5) == 0:
            print(f'  k {k:.1e}: upper row {u[1]:.0f} (x {u[0]:.0f}, {u[2]} lines, rms {u[3]:.1f} px, line rms {u[4]:.2f}), '
                  f'middle row {m[1]:.0f} (x {m[0]:.0f}, {m[2]} lines, rms {m[3]:.1f} px, line rms {m[4]:.2f})')
    # The distortion at which the two tiers' vanishing points agree (their rows cross).
    for (k0, u0, m0), (k1, u1, m1) in zip(rows, rows[1:]):
        d0, d1 = u0[1] - m0[1], u1[1] - m1[1]
        if d0 * d1 <= 0:
            w = d0 / (d0 - d1)
            k = k0 + w * (k1 - k0)
            yv = (u0[1] + w * (u1[1] - u0[1]) + m0[1] + w * (m1[1] - m0[1])) / 2
            xv = (u0[0] + w * (u1[0] - u0[0]) + m0[0] + w * (m1[0] - m0[0])) / 2
            return k, yv, xv
    raise ValueError('the two tiers never agree on the grid')

def pitch(f, yv, cy):
    return math.atan(f / (cy - yv))


def t_centre(v, f=P8_F, yv=None, k=None, cy=P8_CY):
    """Tangent of a ray's elevation, for an image row on the centre column (where the roll does not enter)."""
    yv = P8_YV if yv is None else yv
    k = P8_K if k is None else k
    th = pitch(f, yv, cy)
    vu = undistort(P8_CX, v, k, P8_CX, cy)[1]
    return math.tan(th + math.atan((cy - vu) / f))


def unroll(du, dv, xv, yv, cx, cy):
    """Rotate image offsets by -phi, phi the vertical vanishing point's sideways lean, so that point lands on the centre column."""
    phi = math.atan2(xv - cx, cy - yv)
    return du * math.cos(phi) + dv * math.sin(phi), -du * math.sin(phi) + dv * math.cos(phi)


def unroll_old(du, dv, xv, yv, cx, cy):
    """The rotation review 0 found reversed (F0), kept only to show the check below fails on it."""
    phi = math.atan2(xv - cx, cy - yv)
    return du * math.cos(phi) - dv * math.sin(phi), du * math.sin(phi) + dv * math.cos(phi)


def level(u, v, f=P8_F, yv=None, xv=None, k=None, cx=P8_CX, cy=P8_CY, rot=unroll):
    """(s, t): a ray's lateral and elevation tangents in the level frame looking along the hall, for any image point."""
    yv = P8_YV if yv is None else yv
    xv = P8_XV if xv is None else xv
    k = P8_K if k is None else k
    ux, uy = undistort(u, v, k, cx, cy)
    du2, dv2 = rot(ux - cx, uy - cy, xv, yv, cx, cy)
    th = math.atan(f / math.hypot(xv - cx, cy - yv))
    up, fwd = -dv2, f
    wy = up * math.cos(th) + fwd * math.sin(th)
    wz = -up * math.sin(th) + fwd * math.cos(th)
    return du2 / wz, wy / wz


def vertical_check(rot):
    """A vertical line in the hall must keep one lateral tangent at every height: feed level() points on the line through the vanishing point."""
    pts = [(P8_XV + (y - P8_YV) * (500 - P8_XV) / (2400 - P8_YV), y) for y in (600, 1500, 2400)]   # the image line from the VP through (500, 2400)
    ss = [level(u, v, k=0, rot=rot)[0] for u, v in pts]
    return max(ss) - min(ss)


assert vertical_check(unroll) < 1e-9, 'level() does not keep a vertical line vertical'
assert vertical_check(unroll_old) > 1e-3, 'the check would pass the reversed rotation too'


def fit_screen(f=P8_F, yv=None, k=None, cy=P8_CY, scale=1.0):
    """The friezes' heights (times the screen's scale factor) against their tangents: h = eye + distance * t."""
    pts = [(t_centre(v, f, yv, k, cy), h * scale) for v, h in FRIEZES]
    n = len(pts)
    st = sum(p[0] for p in pts); sh = sum(p[1] for p in pts)
    stt = sum(p[0] ** 2 for p in pts); sth = sum(p[0] * p[1] for p in pts)
    ds = (n * sth - st * sh) / (n * stt - st * st)
    hc = (sh - ds * st) / n
    rms = math.sqrt(sum((h - (hc + ds * t)) ** 2 for t, h in pts) / n)
    return ds, hc, rms


def p8(f=P8_F, yv=None, k=None, D=D_SCREEN_EAST, cy=P8_CY, scale=1.0, setback=0.37):
    yv = P8_YV if yv is None else yv
    k = P8_K if k is None else k
    ds, hc, rms = fit_screen(f, yv, k, cy, scale)
    dw = ds + D
    t_top, t_bot = t_centre(RING_TOP, f, yv, k, cy), t_centre(RING_BOT, f, yv, k, cy)
    ring_centre = hc + dw * (t_top + t_bot) / 2
    ring_v = dw * (t_top - t_bot)
    dg = dw + setback
    g_top, g_bot = t_centre(GLAZ_TOP, f, yv, k, cy), t_centre(GLAZ_BOT, f, yv, k, cy)
    glaz_centre = hc + dg * (g_top + g_bot) / 2
    glaz_v = dg * (g_top - g_bot)
    th = pitch(f, yv, cy)
    a = th + math.atan((cy - undistort(P8_CX, GLAZ_ROW, k, P8_CX, cy)[1]) / f)
    glaz_h = GLAZ_W * dg * math.cos(a - th) / math.cos(a) / f
    ring_h = RING_W * dw * math.cos(a - th) / math.cos(a) / f
    return dict(ds=ds, hc=hc, rms=rms, dw=dw, pitch=math.degrees(th), ring_centre=ring_centre, ring_v=ring_v, ring_h=ring_h,
                glaz_centre=glaz_centre, glaz_v=glaz_v, glaz_h=glaz_h)


def p8_setback():
    lo, hi = -0.5, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2
        r = p8(setback=mid)
        lo, hi = (mid, hi) if r['glaz_centre'] < r['ring_centre'] else (lo, mid)
    return (lo + hi) / 2

def arcade(pts):
    st_ = [level(u, v) for u, v in pts]
    inv = [1 / abs(s) for s, _ in st_]
    mi = sum(inv) / 3
    slope = sum((j - 1) * (b - mi) for j, b in enumerate(inv)) / 2
    icpt = mi - slope
    X = 5.65 / slope
    return [X * (icpt + slope * j) for j in range(3)], [nom['hc'] + t * X * (icpt + slope * j) for j, (_, t) in enumerate(st_)], X

# =====================================================================================================================================
# #29: a view camera, nearly level, its view shifted up. Work unit 4 measured the screen's 2,790 px from floor (row 3,452) to cross (row 662)
# on the axis and a horizontal scale that grows 0.8 % for every 1,000 px down the image, its vertical vanishing point about 125,000 px above.
# Heights on the screen's plane then follow the 1D projective map h(y) = A (1/(y - yv) - 1/(3452 - yv)); the horizontal scale is
# proportional to (y - yv). The Met gives the screen's size rounded to the foot, so its scale carries about 1 %.
# =====================================================================================================================================
YV29 = 2057.0 - 1000.0 / 0.008     # the vertical vanishing point's row, from the 0.8 % per 1,000 px gradient about row 2,057
Y_FLOOR, Y_CROSS = 3452.0, 662.0


def h_screen(y, scale=1.0):
    A = 15.85 * scale / (1 / (Y_CROSS - YV29) - 1 / (Y_FLOOR - YV29))
    return A * (1 / (y - YV29) - 1 / (Y_FLOOR - YV29))


def sy_at(y, scale=1.0):
    """Vertical scale, px per metre on the screen's plane, at row y."""
    A = 15.85 * scale / (1 / (Y_CROSS - YV29) - 1 / (Y_FLOOR - YV29))
    return (y - YV29) ** 2 / A


def sx_at(y, scale=1.0):
    """Horizontal scale, px per metre on the screen's plane: proportional to (y - yv), equal to the vertical scale at the screen's mid-height."""
    ym = (Y_FLOOR + Y_CROSS) / 2
    return sy_at(ym, scale) * (y - YV29) / (ym - YV29)


FRAME_W = 2670.0          # the east doorway's intrados: jambs at x 249 and 2,919 (rows 2,100-3,300), crown at row 382


def circle_fit(pts):
    """Algebraic (Kasa) circle fit: centre x, centre y, radius, rms of the radial residuals, residuals."""
    n = len(pts)
    mx = sum(p[0] for p in pts) / n; my = sum(p[1] for p in pts) / n
    suu = sum((p[0] - mx) ** 2 for p in pts); svv = sum((p[1] - my) ** 2 for p in pts); suv = sum((p[0] - mx) * (p[1] - my) for p in pts)
    suuu = sum((p[0] - mx) ** 3 for p in pts); svvv = sum((p[1] - my) ** 3 for p in pts)
    suvv = sum((p[0] - mx) * (p[1] - my) ** 2 for p in pts); svuu = sum((p[1] - my) * (p[0] - mx) ** 2 for p in pts)
    det = suu * svv - suv * suv
    b0, b1 = (suuu + suvv) / 2, (svvv + svuu) / 2
    uc, vc = (b0 * svv - suv * b1) / det, (suu * b1 - suv * b0) / det
    cx, cy = uc + mx, vc + my
    r = math.sqrt(uc * uc + vc * vc + (suu + svv) / n)
    res = [math.hypot(p[0] - cx, p[1] - cy) - r for p in pts]
    return cx, cy, r, math.sqrt(sum(e * e for e in res) / n), res


def window_circles(path):
    """#29's west window from its edge readings: the surround's edge, and the glazing's iron frame (mid line, inner and outer edges)."""
    S, F = [], []
    for line in open(path, encoding='utf-8'):
        if line.startswith('S '):
            S.append([float(v) for v in line.split()[1:]])
        elif line.startswith('F '):
            F.append([None if v == 'NA' else float(v) for v in line.split()[1:]])
    # The surround: keep rays whose two plateaus are clean (the grey splay against the lighter wall), then drop residuals over 2 px.
    s_ok = [r for r in S if r[4] <= 16 and r[5] <= 12 and 110 < r[2] < 160 and r[3] > 160 and 200 < r[1] < 225]
    fit = circle_fit([(r[6], r[7]) for r in s_ok])
    s_keep = [r for r, e in zip(s_ok, fit[4]) if abs(e) < 2.0]
    ring = circle_fit([(r[6], r[7]) for r in s_keep])
    # The frame: rays that cross a dark ring with bright glazing inside it, measured from the surround's centre as read on the image.
    ocx, ocy = 1596.5, 1323.4
    xy = (lambda ang, r: (ocx + r * math.cos(math.radians(ang)), ocy - r * math.sin(math.radians(ang))))
    f_ok = [r for r in F if r[2] < 70 and r[3] > 160 and r[5] is not None and r[6] is not None]
    fit = circle_fit([xy(r[0], (r[5] + r[6]) / 2) for r in f_ok])
    f_keep = [r for r, e in zip(f_ok, fit[4]) if abs(e) < 2.5]
    mid = circle_fit([xy(r[0], (r[5] + r[6]) / 2) for r in f_keep])
    inner = circle_fit([xy(r[0], r[5]) for r in f_keep])
    outer = circle_fit([xy(r[0], r[6]) for r in f_keep])
    print(f'#29 surround: {len(s_keep)} points, centre ({ring[0]:.1f}, {ring[1]:.1f}), radius {ring[2]:.1f} px, rms {ring[3]:.2f} px')
    print(f'#29 frame: {len(f_keep)} points, centre ({mid[0]:.1f}, {mid[1]:.1f}), radius {mid[2]:.1f} px (inner {inner[2]:.1f}, outer {outer[2]:.1f}), rms {mid[3]:.2f} px')
    return (ring[0], ring[1], ring[2]), (mid[0], mid[1], mid[2], inner[2], outer[2])

def m29(door=DOOR, d_es=D_EAST_TO_SCREEN, d_sw=D_SCREEN_TO_WEST, base_off=0.15, thr=3299.0, base=3452.5, scale=1.0):
    s_frame = sx_at(2700, scale)                  # the framing doorway is read about row 2,700
    d_east = s_frame * d_es / (FRAME_W / door - s_frame)    # FRAME_W = f door / d_east, with f = s ds and ds = d_east + d_es
    ds = d_east + d_es
    f = s_frame * ds
    dw = ds + d_sw
    r = dw / ds
    # The base row (where the floor meets the base's front face, base_off nearer than the screen's plane) stands at
    # delta = -hc base_off / (ds - base_off) on the screen's plane; heights read from that row are shifted by delta.
    h_thr = h_screen(thr, scale) - h_screen(base, scale)
    hc, delta = 2.8, 0.0
    for _ in range(50):
        delta = -hc * base_off / (ds - base_off)
        hc = (h_thr + delta) * dw / (dw - ds)      # the ray through the threshold's row meets the floor at dw
    ring_c = h_screen(RING[1], scale) - h_screen(base, scale) + delta
    hw = hc + (ring_c - hc) * r
    ring_v = (h_screen(RING[1] - RING[2], scale) - h_screen(RING[1] + RING[2], scale)) * r
    ring_h = 2 * RING[2] / sx_at(RING[1], scale) * r

    west_near = f * door / dw
    west_far = f * door / (dw + 1.15)
    return dict(d_east=d_east, ds=ds, f=f, dw=dw, r=r, hc=hc, ring_c_app=ring_c, hw=hw, ring_v=ring_v, ring_h=ring_h, gl_out=gl_out, gl_in=gl_in,

def crossing(key):
    lo, hi = -3.0, 3.0
    f0 = (lambda d: both(**{key: d})[0] - both(**{key: d})[1])
    s_lo = f0(lo) > 0
    for _ in range(60):
        mid = (lo + hi) / 2
        lo, hi = (mid, hi) if (f0(mid) > 0) == s_lo else (lo, mid)
    e, w_, _ = both(**{key: (lo + hi) / 2})
    return (lo + hi) / 2, e, w_
