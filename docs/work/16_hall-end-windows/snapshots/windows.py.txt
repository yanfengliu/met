def find_repo():
    for start in (os.environ.get('MET_REPO', ''), os.path.dirname(os.path.abspath(__file__)), os.getcwd()):
        d = os.path.abspath(start) if start else ''
        while d:
            if all(os.path.exists(os.path.join(d, 'src', 'model', *p)) for p in (('core', 'constants.ts'), ('frame.ts',))):
                return d
            d = os.path.dirname(d) if os.path.dirname(d) != d else ''
    return None


def read_model():
    repo = find_repo()
    if repo is None:
        return dict(MODEL_COPIES), "this script's own copies (the repository was not found, so no model is checked)"
    found = {}
    for rel, names in (('src/model/core/constants.ts', ('VAUX_OCULUS_Y', 'VAUX_OCULUS')), ('src/model/frame.ts', ('MAIN_FLOOR_Y',))):
        text = open(os.path.join(repo, *rel.split('/')), encoding='utf-8').read()
        for name in names:
            m = re.search(r'^export const ' + name + r' = ([0-9.]+);', text, re.M)
            assert m, f'{name} is not a number literal in {rel}'
            found[name] = float(m.group(1))
    return found, "the repository's src/model/core/constants.ts and src/model/frame.ts"

def undistort(x, y, k, cx=P8_CX, cy=P8_CY):
    dx, dy = x - cx, y - cy
    s = 1 + k * (dx * dx + dy * dy)
    return cx + dx * s, cy + dy * s


# ---- The balusters' vertical vanishing point, refitted on distortion-corrected positions -----------------------------------------------
def load_balusters(path):
    tiers = {'U': [], 'M': []}
    for line in open(path, encoding='utf-8'):
        if not line.strip() or line.startswith('#'):
            continue
        t, i, x, y = line.split()
        tiers[t].append((int(i), float(x), float(y)))
    return tiers


def vanishing_point(samples, k, yref, cx=P8_CX, cy=P8_CY):
    """Fit a line x = a + b (y - yref) through each baluster's corrected centres, then the point they meet: a_i = X - (Y - yref) b_i."""
    by = {}
    for i, x, y in samples:
        by.setdefault(i, []).append(undistort(x, y, k, cx, cy))
    lines = []
    for i, pts in by.items():
        if len(pts) < 20:
            continue
        n = len(pts)
        sy = sum(p[1] - yref for p in pts); sx = sum(p[0] for p in pts)
        syy = sum((p[1] - yref) ** 2 for p in pts); sxy = sum(p[0] * (p[1] - yref) for p in pts)
        b = (n * sxy - sy * sx) / (n * syy - sy * sy)
        a = (sx - b * sy) / n
        rms = math.sqrt(sum((p[0] - (a + b * (p[1] - yref))) ** 2 for p in pts) / n)
        if rms < 2.5:
            lines.append((a, b, rms, i))
    m = len(lines)
    sb = sum(l[1] for l in lines); sa = sum(l[0] for l in lines)
    sbb = sum(l[1] ** 2 for l in lines); sab = sum(l[0] * l[1] for l in lines)
    slope = (m * sab - sb * sa) / (m * sbb - sb * sb)
    X = (sa - slope * sb) / m
    rms = math.sqrt(sum((l[0] - (X + slope * l[1])) ** 2 for l in lines) / m)
    line_rms = math.sqrt(sum(l[2] ** 2 for l in lines) / m)
    return X, yref - slope, m, rms, line_rms, [l[3] for l in lines]


def vp_jackknife(samples, k, yref):
    """The vanishing point's row and its standard error over the balusters: refitted leaving out each baluster in turn."""
    full = vanishing_point(samples, k, yref)
    vals = [vanishing_point([s for s in samples if s[0] != i], k, yref)[1] for i in full[5]]
    m = len(vals)
    mean = sum(vals) / m
    return full[1], math.sqrt((m - 1) / m * sum((v - mean) ** 2 for v in vals))


def refit(path, cx=P8_CX, cy=P8_CY, show=False):
    tiers = load_balusters(path)
    grid = [j * 0.1e-8 for j in range(0, 51)]
    rows = []
    for k in grid:
        u = vanishing_point(tiers['U'], k, 820, cx, cy)
        m = vanishing_point(tiers['M'], k, 1350, cx, cy)
        rows.append((k, u, m))
        if show and abs(round(k * 1e9) % 5) == 0:
            print(f'  k {k:.1e}: upper row {u[1]:.0f} (x {u[0]:.0f}, {u[2]} lines, rms {u[3]:.1f} px, line rms {u[4]:.2f}), '
                  f'middle row {m[1]:.0f} (x {m[0]:.0f}, {m[2]} lines, rms {m[3]:.1f} px, line rms {m[4]:.2f})')
    # The distortion at which the two tiers' vanishing points agree (their rows cross).
    for (k0, u0, m0), (k1, u1, m1) in zip(rows, rows[1:]):
        d0, d1 = u0[1] - m0[1], u1[1] - m1[1]
        if d0 * d1 <= 0:
            w = d0 / (d0 - d1)
            k = k0 + w * (k1 - k0)
            yv = (u0[1] + w * (u1[1] - u0[1]) + m0[1] + w * (m1[1] - m0[1])) / 2
            xv = (u0[0] + w * (u1[0] - u0[0]) + m0[0] + w * (m1[0] - m0[0])) / 2
            return k, yv, xv, (u0[0] + w * (u1[0] - u0[0])), (m0[0] + w * (m1[0] - m0[0])), (d1 - d0) / (k1 - k0)
    raise ValueError('the two tiers never agree on the grid')

# ---- The friezes' edges: straight lines in the hall, so straight once the distortion is undone -------------------------------------------
def load_friezes(path):
    edges = {}
    for line in open(path, encoding='utf-8'):
        if not line.strip() or line.startswith('#'):
            continue
        name, x, y = line.split()
        edges.setdefault(name, []).append((float(x), float(y)))
    return edges


def line_fit(pts):
    n = len(pts)
    sx = sum(p[0] for p in pts); sy = sum(p[1] for p in pts)
    sxx = sum(p[0] ** 2 for p in pts); sxy = sum(p[0] * p[1] for p in pts)
    b = (n * sxy - sx * sy) / (n * sxx - sx * sx)
    a = (sy - b * sx) / n
    return a, b, [p[1] - (a + b * p[0]) for p in pts]


def straight_edges(edges, k_select=2.0e-8):
    """Drop each edge's outliers (a baluster's or an ornament's edge read instead) by iterated 3-sigma clipping, at least 1 px, at k_select;
    then find the distortion that makes the kept readings straightest, all edges together and edge by edge, with the joint value's standard error."""
    kept = {}
    for name, pts in edges.items():
        idx = list(range(len(pts)))
        for _ in range(10):
            _, _, res = line_fit([undistort(pts[i][0], pts[i][1], k_select) for i in idx])
            rms = math.sqrt(sum(r * r for r in res) / len(res))
            new = [i for i, r in zip(idx, res) if abs(r) < max(3 * rms, 1.0)]
            if new == idx:
                break
            idx = new
        kept[name] = [pts[i] for i in idx]
    cost = (lambda pts, k: sum(r * r for r in line_fit([undistort(x, y, k) for x, y in pts])[2]))
    grid = [j * 0.1e-8 for j in range(0, 61)]

    def best(f, n_points, n_params):
        """The minimum of f over the grid, refined by the parabola through its three lowest points, and its standard error from the
        parabola's curvature and the residual variance (n_points readings, n_params fitted)."""
        j = min(range(1, len(grid) - 1), key=lambda i: f(grid[i]))
        c0, c1, c2, h = f(grid[j - 1]), f(grid[j]), f(grid[j + 1]), grid[1]
        k = grid[j] + h * (c0 - c2) / (2 * (c0 - 2 * c1 + c2))
        s2 = f(k) / (n_points - n_params)
        return k, math.sqrt(2 * s2 / ((c0 - 2 * c1 + c2) / h ** 2))
    n = sum(len(p) for p in kept.values())
    k_joint, se = best(lambda k: sum(cost(p, k) for p in kept.values()), n, 2 * len(kept) + 1)   # two parameters per edge, and the distortion
    each = {name: best(lambda k, p=p: cost(p, k), len(p), 3) for name, p in kept.items()}
    rms = {name: math.sqrt(cost(p, k_joint) / len(p)) for name, p in kept.items()}
    rms0 = {name: math.sqrt(cost(p, 0.0) / len(p)) for name, p in kept.items()}
    return k_joint, se, each, rms, rms0, {name: (len(kept[name]), len(edges[name])) for name in edges}

def pitch(f, yv, cy):
    return math.atan(f / (cy - yv))


def t_centre(v, f=P8_F, yv=None, k=None, cy=P8_CY):
    """Tangent of a ray's elevation, for an image row on the centre column (where the roll does not enter)."""
    yv = P8_YV if yv is None else yv
    k = P8_K if k is None else k
    th = pitch(f, yv, cy)
    vu = undistort(P8_CX, v, k, P8_CX, cy)[1]
    return math.tan(th + math.atan((cy - vu) / f))


def unroll(du, dv, xv, yv, cx, cy):
    """Rotate image offsets by -phi, phi the vertical vanishing point's sideways lean, so that point lands on the centre column."""
    phi = math.atan2(xv - cx, cy - yv)
    return du * math.cos(phi) + dv * math.sin(phi), -du * math.sin(phi) + dv * math.cos(phi)


def unroll_old(du, dv, xv, yv, cx, cy):
    """The rotation review 0 found reversed (F0), kept only to show the check below fails on it."""
    phi = math.atan2(xv - cx, cy - yv)
    return du * math.cos(phi) - dv * math.sin(phi), du * math.sin(phi) + dv * math.cos(phi)


def level(u, v, f=P8_F, yv=None, xv=None, k=None, cx=P8_CX, cy=P8_CY, rot=unroll):
    """(s, t): a ray's lateral and elevation tangents in the level frame looking along the hall, for any image point."""
    yv = P8_YV if yv is None else yv
    xv = P8_XV if xv is None else xv
    k = P8_K if k is None else k
    ux, uy = undistort(u, v, k, cx, cy)
    du2, dv2 = rot(ux - cx, uy - cy, xv, yv, cx, cy)
    th = math.atan(f / math.hypot(xv - cx, cy - yv))
    up, fwd = -dv2, f
    wy = up * math.cos(th) + fwd * math.sin(th)
    wz = -up * math.sin(th) + fwd * math.cos(th)
    return du2 / wz, wy / wz


def vertical_check(rot):
    """A vertical line in the hall must keep one lateral tangent at every height: feed level() points on the line through the vanishing point."""
    pts = [(P8_XV + (y - P8_YV) * (500 - P8_XV) / (2400 - P8_YV), y) for y in (600, 1500, 2400)]   # the image line from the VP through (500, 2400)
    ss = [level(u, v, k=0, rot=rot)[0] for u, v in pts]
    return max(ss) - min(ss)


assert vertical_check(unroll) < 1e-9, 'level() does not keep a vertical line vertical'
assert vertical_check(unroll_old) > 1e-3, 'the check would pass the reversed rotation too'


def fit_screen(f=P8_F, yv=None, k=None, cy=P8_CY, scale=1.0, friezes=FRIEZES):
    """The friezes' heights (times the screen's scale factor) against their tangents: h = eye + distance * t."""
    pts = [(t_centre(v, f, yv, k, cy), h * scale) for v, h in friezes]
    n = len(pts)
    st = sum(p[0] for p in pts); sh = sum(p[1] for p in pts)
    stt = sum(p[0] ** 2 for p in pts); sth = sum(p[0] * p[1] for p in pts)
    ds = (n * sth - st * sh) / (n * stt - st * st)
    hc = (sh - ds * st) / n
    rms = math.sqrt(sum((h - (hc + ds * t)) ** 2 for t, h in pts) / n)
    return ds, hc, rms


def p8(f=P8_F, yv=None, k=None, D=D_SCREEN_EAST, cy=P8_CY, scale=1.0, setback=0.37, friezes=FRIEZES):
    yv = P8_YV if yv is None else yv
    k = P8_K if k is None else k
    ds, hc, rms = fit_screen(f, yv, k, cy, scale, friezes)
    dw = ds + D
    t_top, t_bot = t_centre(RING_TOP, f, yv, k, cy), t_centre(RING_BOT, f, yv, k, cy)
    ring_centre = hc + dw * (t_top + t_bot) / 2
    ring_v = dw * (t_top - t_bot)
    dg = dw + setback
    g_top, g_bot = t_centre(GLAZ_TOP, f, yv, k, cy), t_centre(GLAZ_BOT, f, yv, k, cy)
    glaz_centre = hc + dg * (g_top + g_bot) / 2
    glaz_v = dg * (g_top - g_bot)
    th = pitch(f, yv, cy)
    a = th + math.atan((cy - undistort(P8_CX, GLAZ_ROW, k, P8_CX, cy)[1]) / f)

def circle_fit(pts):
    """Algebraic (Kasa) circle fit: centre x, centre y, radius, rms of the radial residuals, residuals."""
    n = len(pts)
    mx = sum(p[0] for p in pts) / n; my = sum(p[1] for p in pts) / n
    suu = sum((p[0] - mx) ** 2 for p in pts); svv = sum((p[1] - my) ** 2 for p in pts); suv = sum((p[0] - mx) * (p[1] - my) for p in pts)
    suuu = sum((p[0] - mx) ** 3 for p in pts); svvv = sum((p[1] - my) ** 3 for p in pts)
    suvv = sum((p[0] - mx) * (p[1] - my) ** 2 for p in pts); svuu = sum((p[1] - my) * (p[0] - mx) ** 2 for p in pts)
    det = suu * svv - suv * suv
    b0, b1 = (suuu + suvv) / 2, (svvv + svuu) / 2
    uc, vc = (b0 * svv - suv * b1) / det, (suu * b1 - suv * b0) / det
    cx, cy = uc + mx, vc + my
    r = math.sqrt(uc * uc + vc * vc + (suu + svv) / n)
    res = [math.hypot(p[0] - cx, p[1] - cy) - r for p in pts]
    return cx, cy, r, math.sqrt(sum(e * e for e in res) / n), res


def window_circles(path):
    """#29's west window from its edge readings: the surround's edge, and the glazing's iron frame (mid line, inner and outer edges)."""
    S, F = [], []

# =====================================================================================================================================
# #29: a view camera, nearly level, its view shifted up, solved as one pinhole camera. Unknowns: its focal length f, its tilt theta, the
# principal point's row y0, its eye height hc and its distance to the east doorway's west face. Conditions:
#   the cross's tip, 15.85 m up on the screen's line, CROSS_BEHIND beyond its east face, at row 662 (work unit 4);
#   the floor meeting the screen's base, base_off nearer than the screen's east face, at row 3,452.5;
#   the west doorway's threshold, on the floor at the west wall, at row 3,299;
#   the vertical vanishing point at row YV29, from the 0.8 % per 1,000 px growth of the balusters' spacing (work unit 4);
#   the east doorway, door metres wide on the map, 2,670 px across at row 2,700.
# A row y sees along the elevation alpha(y) = theta + atan((y0 - y) / f); a point at height h and horizontal distance d lands at the row where
# tan alpha = (h - hc) / d, and its horizontal scale there is f / z, z = d cos(alpha - theta) / cos(alpha).
# =====================================================================================================================================
YV29 = 2057.0 - 1000.0 / 0.008     # -122,943: the vertical vanishing point's row
Y_CROSS, Y_BASE = 662.0, 3452.5
CROSS_BEHIND = Z_SCREEN_FRONT_E - Z_PARTITION   # 0.10: the cresting and its cross stand on the screen's line, as the model builds them
FRAME_W, FRAME_ROW = 2670.0, 2700.0


def alpha(y, f, y0):
    """Elevation of the ray through row y: the tilt follows from the vanishing point, tan(theta) = f / (y0 - YV29)."""
    th = math.atan(f / (y0 - YV29))
    return th + math.atan((y0 - y) / f)


def tilt(f, y0):
    return math.atan(f / (y0 - YV29))


def m29_solve(door=DOOR, d_es=D_EAST_TO_SCREEN, d_sw=D_SCREEN_TO_WEST, base_off=0.15, thr=3299.0, scale=1.0):
    """Solve (f, y0) by Newton's method on the cross and the framing doorway; the tilt follows from the vanishing point, the rest in closed form."""
    def rest(f, y0):
        th = tilt(f, y0)
        tb, tt = math.tan(alpha(Y_BASE, f, y0)), math.tan(alpha(thr, f, y0))
        ds = (d_sw * tt + base_off * tb) / (tb - tt)       # (ds - base_off) tan(a_base) = (ds + d_sw) tan(a_thr): both rays meet the floor
        hc = -(ds - base_off) * tb
        de = ds - d_es
        ae = alpha(FRAME_ROW, f, y0)
        r1 = math.tan(alpha(Y_CROSS, f, y0)) - (15.85 * scale - hc) / (ds + CROSS_BEHIND)
        r2 = (FRAME_W - f * door * math.cos(ae) / (de * math.cos(ae - th))) / FRAME_W
        return r1, r2, ds, hc, de
    f, y0 = 4730.0, 2760.0
    for _ in range(200):
        r1, r2, *_ = rest(f, y0)
        h = 0.5
        a1, a2, *_ = rest(f + h, y0)
        b1, b2, *_ = rest(f, y0 + h)
        j11, j12, j21, j22 = (a1 - r1) / h, (b1 - r1) / h, (a2 - r2) / h, (b2 - r2) / h
        det = j11 * j22 - j12 * j21
        df = (r1 * j22 - r2 * j12) / det
        dy = (j11 * r2 - j21 * r1) / det
        step, nf, ny = 1.0, f, y0
        while step > 1e-3:                                # damped: never step to a worse pair
            nf, ny = f - step * df, y0 - step * dy
            try:
                n1, n2, *_ = rest(nf, ny)
                if n1 * n1 + n2 * n2 < r1 * r1 + r2 * r2:
                    break
            except (ValueError, ZeroDivisionError):
                pass
            step /= 2
        f, y0 = nf, ny
        if abs(step * df) < 1e-6 and abs(step * dy) < 1e-6:
            break
    r1, r2, ds, hc, de = rest(f, y0)
    assert abs(r1) < 1e-10 and abs(r2) < 1e-10, ('the #29 camera did not solve', r1, r2)
    return f, y0, ds, hc, de


def m29(door=DOOR, d_es=D_EAST_TO_SCREEN, d_sw=D_SCREEN_TO_WEST, base_off=0.15, thr=3299.0, scale=1.0):
    f, y0, ds, hc, de = m29_solve(door, d_es, d_sw, base_off, thr, scale)
    th = tilt(f, y0)
    dw = ds + d_sw
    height = (lambda y, d: hc + d * math.tan(alpha(y, f, y0)))
    # The circles are fitted on the wall, in metres: each edge point is carried through the camera onto a wall at unit distance, lateral
    # (x - X0_29) cos(a - th) / (f cos a) and height over the eye tan a, a = alpha(y). At distance d both scale by d, and an algebraic circle
    # fit follows a uniform scaling, so one fit per circle serves every distance: centre height hc + d * cy1, radius d * r1.
    def unit_fit(points):
        wall = [((x - X0_29) * math.cos(alpha(y, f, y0) - th) / (math.cos(alpha(y, f, y0)) * f), math.tan(alpha(y, f, y0))) for x, y in points]
        return circle_fit(wall)
    s_fit, m_fit, i_fit, o_fit = (unit_fit(EDGE_PTS[k]) for k in ('surround', 'mid', 'inner', 'outer'))
    hw = hc + dw * s_fit[1]
    ring = 2 * dw * s_fit[2]
    # The glazing's set-back: where the frame's centre stands as high as the surround's.
    dg = (hw - hc) / m_fit[1]
    setback = dg - dw
    gl = 2 * dg * o_fit[2]
    gl_in = 2 * dg * i_fit[2]
    a_thr = alpha(3150, f, y0)
    west_near = f * door * math.cos(a_thr) / (dw * math.cos(a_thr - th))
    west_far = f * door * math.cos(a_thr) / ((dw + 1.15) * math.cos(a_thr - th))
    cross_check = height(Y_CROSS, ds + CROSS_BEHIND)
    return dict(f=f, tilt=math.degrees(th), y0=y0, d_east=de, ds=ds, dw=dw, hc=hc, hw=hw, ring=ring, gl=gl, gl_in=gl_in, setback=setback,
                ring_rms_mm=1000 * dw * s_fit[3], frame_rms_mm=1000 * dg * m_fit[3], west_near_px=west_near, west_far_px=west_far,
                cross=cross_check, cam_z=Z_EAST_DOOR_W_FACE + de)

def glazing(p):
    """The glazing's weighted average with P8's two readings taken from p8()'s result p and #29's nominal reading, and its rounding."""
    g = ((p['glaz_v'] + p['glaz_h']) / 2 / g8h ** 2 + n29['gl'] / g29h ** 2) / (1 / g8h ** 2 + 1 / g29h ** 2)
    r, margin = rounded(g, 0.05)
    return f'the glazing {g:.3f} m, which rounds to {r:.2f}, {margin:.0f} mm inside'
